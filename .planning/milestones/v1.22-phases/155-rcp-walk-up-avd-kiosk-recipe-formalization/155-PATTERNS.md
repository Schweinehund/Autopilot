**Parser this canary tests, for reference** (`:103-114`, do not modify — it is generic over row count):
```js
export function parseRegistry(content) {
  return content
    .split(/\r?\n/)
    .filter(l => /^\|\s*RE-\d+\s*\|/.test(l))
    .map(l => {
      const cols = l.split('|').map(s => s.trim());
      return { docId: cols[1], path: cols[2], title: cols[3], docType: cols[4], status: cols[5] };
    });
}
```
